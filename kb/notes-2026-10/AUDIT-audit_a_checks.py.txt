# (c) formule R independante (a la main) : D = dL - dL - dphi L + ... , R = Omega^-1 D
def Rhand(Y):
    trT = trace(Y)
    eta = sp.diag(-1, 1, 1, 1)
    L = ((kap/2)*(Y - sp.Rational(1, 3)*trT*tau**-2*eta)).applyfunc(sp.expand)
    dphi = [-1/tau, 0, 0, 0]; dphiU = [1/tau, 0, 0, 0]
    def d(mu, e): return [M['Dt'](e), I*k*e, 0, 0][mu]
    out = {}
    for r in range(4):
        for n in range(4):
            for l in range(n+1, 4):
                D = (d(n, L[l, r]) - d(l, L[n, r]) - dphi[n]*L[l, r] + dphi[l]*L[n, r]
                     + eta[n, r]*sum(dphiU[s]*L[l, s] for s in range(4))
                     - eta[l, r]*sum(dphiU[s]*L[n, s] for s in range(4)))
                out[(r, n, l)] = sp.expand(D/(-tau))
    return out
hf = sp.Function('h')(tau)
Ya = sol_from_h(sum(sp.Symbol(f'q{n}{j}')*tau**n*Lg**j for n in range(-2, 5) for j in range(3)))
R1, R2 = Rsrc(Ya), Rhand(Ya)
print("(c) Rsrc == formule manuelle :", all(sp.expand(R1[x]-R2[x]) == 0 for x in R1))

# (d) verification avec vrai ln(-tau) et diff sympy (pas l'algebre Lg)
Ys = M['Yreg'].subs({sp.Symbol(s): sp.Rational(v) for s, v in [('a_1_0', 1), ('a_2_0', 2), ('a_3_0', -3), ('a_4_0', 5), ('a_5_0', 7)]})
Ys = Ys.applyfunc(lambda e: e.subs({sym: 0 for sym in e.free_symbols if str(sym).startswith('a_')}).subs(Lg, sp.log(-tau)))
Y00, Y0x, Yxx, Yyy = Ys[0, 0], Ys[0, 1], Ys[1, 1], Ys[2, 2]
c0 = sp.simplify(-sp.diff(Y00, tau) + I*k*Y0x + 2*Y00/tau + A*tau)
c1 = sp.simplify(-sp.diff(Y0x, tau) + I*k*Yxx + 2*Y0x/tau)
ctr = sp.simplify(-Y00 + Yxx + 2*Yyy - A*tau**2)
print("(d) conservation/trace avec vrai log :", c0, c1, ctr)
