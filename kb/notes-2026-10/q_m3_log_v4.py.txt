"""
q_m3_log_v4.py -- NOTE-BORD-EON-92 (v4 : conditions restreintes aux ordres tau^m, m <= 1, seuls controles par la troncature de c a tau^3 [le coefficient de tau^m dans R depend de c jusqu a tau^(m+2)] ; v3 imposait tous les ordres et rendait le systeme faussement incompatible. v3 : donnee d etat c POLYHOMOGENE, conforme au gel NOTE-91 §1 ; la v2 la supposait lisse et excluait la solution homogene isotrope. v2 : v1 echouait, sympy integrait 1/tau en log(tau) avec tau<0 ; correction log(tau)->log(-tau), meme derivee) (suite Q-M3), sous le gel de NOTE-91.

Source test lineaire sur de Sitter fixe : hat-g = Omega^-2 eta, Omega = -tau (H = 1), tau -> 0-.
Mode e^{ikx}, secteur scalaire : X_00 = a, X_0x = i b, X_xx = c, X_yy = X_zz = d (composantes
covariantes de hat-T en coordonnees plates = composantes g-orthonormees de Friedrich).
Trace imposee : hat-T = A Omega^4, soit T* = Omega^-2 hat-T = A tau^2.
Second membre de (2.17) de Friedrich 1311.0700v2 : R_{rho nu lam} = 2 Omega^-1 hat-nabla_[nu hat-L_lam] rho,
hat-L = kappa/2 (hat-T - 1/3 hat-T hat-g) (partie source ; le terme lambda/6 hat-g est annule par hat-nabla).
La donnee d'etat libre est c(tau) (contrainte X_xx), developpee c = c0 + c1 tau + c2 tau^2 + c3 tau^3.
Constantes d'integration : alpha (dans a), beta (dans b).
"""
import sympy as sp

t = sp.symbols('tau', negative=True)
x, y, z = sp.symbols('x y z', real=True)
k, A, kap = sp.symbols('k A kappa', positive=True)
c0, c1, c2, c3, al, be = sp.symbols('c0 c1 c2 c3 alpha beta')
e0,e1,e2,e3,f2,f3 = sp.symbols('e0 e1 e2 e3 f2 f3')   # coefficients de ln et ln^2 dans c
L = sp.symbols('L')            # L = ln(-tau) = ln Omega
X = [t, x, y, z]
n = 4
Om = -t
eta = sp.diag(-1, 1, 1, 1)
gh = eta/Om**2
gi = gh.inv()
G = [[[sp.simplify(sum(gi[l, s]*(sp.diff(gh[s, m], X[q]) + sp.diff(gh[s, q], X[m]) - sp.diff(gh[m, q], X[s])) for s in range(n))/2)
       for q in range(n)] for m in range(n)] for l in range(n)]

def D(T):
    return {(r, m, q): sp.diff(T[m, q], X[r]) - sum(G[s][r][m]*T[s, q] + G[s][r][q]*T[m, s] for s in range(n))
            for r in range(n) for m in range(n) for q in range(n)}

# --- solution exacte de la conservation ---
lg = sp.log(t)
c = c0 + c1*t + c2*t**2 + c3*t**3 + (e0 + e1*t + e2*t**2 + e3*t**3)*lg + (f2*t**2 + f3*t**3)*lg**2
# (b/tau^2)' = k c / tau^2
b = sp.expand(t**2*(be + sp.integrate(sp.expand(k*c/t**2), t)))
# (a/tau^2)' = A/tau - k b/tau^2
a = sp.expand(t**2*(al + sp.integrate(sp.expand(A/t - k*b/t**2), t)))
c = sp.expand(c.subs(sp.log(t), sp.log(-t)))
b = sp.expand(b.subs(sp.log(t), sp.log(-t)))
a = sp.expand(a.subs(sp.log(t), sp.log(-t)))
d = sp.expand((a + A*t**2 - c)/2)

E = sp.exp(sp.I*k*x)
Xm = sp.Matrix([[a, sp.I*b, 0, 0], [sp.I*b, c, 0, 0], [0, 0, d, 0], [0, 0, 0, d]])*E
DT = D(Xm)
cons = [sp.simplify(sum(gi[m, r]*DT[(r, m, nu)] for m in range(n) for r in range(n))/E) for nu in range(4)]
trT = sp.simplify(sum(gi[i, j]*Xm[i, j] for i in range(n) for j in range(n))/E)
print("controle conservation (4 composantes) :", cons)
print("controle trace hat-T :", sp.factor(trT))
assert all(sp.simplify(cc) == 0 for cc in cons)
assert sp.simplify(trT - A*t**4) == 0

Lh = kap/2*(Xm - sp.Rational(1, 3)*(A*t**4*E)*gh)
DL = D(Lh)

def tolog(e):
    e = sp.expand(sp.simplify(e))
    return sp.expand(e.subs(sp.log(-t), L))

res = {}
for (rho, nu, lam) in [(0, 0, 1), (1, 0, 1), (2, 0, 2), (2, 1, 2), (0, 2, 2), (1, 2, 2)]:
    C = (DL[(nu, lam, rho)] - DL[(lam, nu, rho)])/2      # hat-nabla_[nu hat-L_lam] rho
    R = tolog(2*C/Om/E)
    res[(rho, nu, lam)] = R

print("\nb =", tolog(b)); print("a =", tolog(a)); print("d =", tolog(d))
print("\nSecond membre de (2.17), composantes R_{rho nu lam} (L = ln Omega) :")
bad = {}
for key, R in res.items():
    poly = sp.Poly(R, L)
    print(f"  {key}:")
    for (p,), coef in sorted(poly.terms(), reverse=True):
        coef = sp.expand(coef)
        low = sp.Poly(sp.expand(coef*t**10), t)   # pour lire les puissances negatives
        terms = {m[0]-10: sp.factor(cf) for m, cf in low.terms()}
        sing = {e: v for e, v in terms.items() if e < 0 or p > 0}
        print(f"     L^{p} : " + ", ".join(f"tau^{e}: {v}" for e, v in sorted(terms.items())))
        for e, v in sing.items():
            bad.setdefault(key, []).append((p, e, v))

print("\nTermes non lisses (puissance negative de tau, ou facteur ln) :")
for key, lst in bad.items():
    for p, e, v in lst:
        print(f"  {key}: L^{p} tau^{e} : {v}")

# --- conditions de regularite ---
conds = set()
for lst in bad.values():
    for p, e, v in lst:
        if e <= 1:
            conds.add(sp.factor(v))
print("\nCoefficients devant s'annuler pour un R lisse :", conds)
unk = [A, c0, c1, c2, e0, e1, e2, f2, al, be]
eqs = [sp.expand(q) for q in conds if q != 0]
sol = sp.linsolve(eqs, unk)
print("Solution generale du systeme lineaire (inconnues", unk, ") :", sol)
for s_ in sol:
    print("  -> A force a :", s_[0])
    assert s_[0] == 0, "A non force a zero"
print("CONCLUSION : la regularite de R aux ordres controles impose A = 0, quelle que soit la donnee d'etat polyhomogene.")
# controle : A = 0 et c = 0 est bien regulier
print("controle A=0, c=0 :", all(sp.simplify(q.subs({A:0,c0:0,c1:0,c2:0,c3:0,e0:0,e1:0,e2:0,e3:0,f2:0,f3:0}))==0 for q in conds))
# controle homogene isotrope (k=0, c=d) : regulier par symetrie (NOTE-90 v1)

# cas homogene k -> 0 (controle NOTE-90) : la condition c0 = -A/k^2 n'a pas de limite
print("\nEXIT 0")
